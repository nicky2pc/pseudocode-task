ИНИЦИАЛИЗАЦИЯ:
  ЦО        <- входной параметр
  дельта    <- входной параметр
  созданные <- пустое множество
  пред_цена <- текущая_цена()

ЦИКЛ (бесконечно):
  цена <- текущая_цена()

  n <- 0
  ЦИКЛ пока (ЦО + n * дельта) <= цена:
    триггер        <- ЦО + n * дельта
    уровень_ордера <- ЦО + (n + 1) * дельта

    ЕСЛИ пред_цена < триггер И цена >= триггер:
      ЕСЛИ уровень_ордера НЕ входит в созданные:
        создать_ордер(ЛОНГ, цена = уровень_ордера)
        создать_ордер(ШОРТ, цена = уровень_ордера)
        добавить уровень_ордера в созданные

    n <- n + 1

  пред_цена <- цена
  подождать(тик)
